cmake_minimum_required(VERSION 3.20)
project(lob_execution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(lob_engine cpp/src/order_book.cpp)
target_include_directories(lob_engine PUBLIC cpp/include)
set_target_properties(lob_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(lob_engine PRIVATE /W4)
else()
    target_compile_options(lob_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(lob_engine_test cpp/tests/order_book_test.cpp)
    target_link_libraries(lob_engine_test PRIVATE lob_engine)
    add_test(NAME lob_engine_test COMMAND lob_engine_test)
endif()
option(LOB_PYTHON "Build Python bindings" ON)
if(LOB_PYTHON)
    find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
                    OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE
                    COMMAND_ERROR_IS_FATAL ANY)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(_lob_cpp cpp/src/bindings.cpp)
    target_link_libraries(_lob_cpp PRIVATE lob_engine)
endif()
